def recording_signals(rec: dict) -> dict:
    """Every cleaned signal of one recording as plain value arrays, plus the rates and QC.

    "signals": 1-D arrays without NaNs, one value per sample (pupil mm, gaze_x/gaze_y deg,
    gaze_speed deg/s) or per event (fixation/saccade/blink durations in ms, saccade amplitude
    deg, saccade velocities deg/s). These are what flatten.py summarizes.
    "rates": blink, fixation and saccade rates. "qc": quality numbers, never model inputs.
    """
    rec = trim_to_gaze_range(rec)
    gaze = rec["gaze"]
    t = gaze["gaze_timestamp"].to_numpy(float)
    vt = valid_time_s(t)
    blinks = clean_blinks(rec["blinks"])

    grid, xyz = gaze_directions(gaze, blinks)
    ev = detect_events(xyz, GAZE_GRID_HZ)
    finite = np.isfinite(xyz).all(1)
    keep = finite & ~ev["artifact_mask"]
    az = np.degrees(np.arctan2(xyz[keep, 0], xyz[keep, 2]))
    el = np.degrees(np.arctan2(xyz[keep, 1], np.hypot(xyz[keep, 0], xyz[keep, 2])))
    speed = angular_speed_deg_s(xyz, GAZE_GRID_HZ)
    speed = speed[np.isfinite(speed) & ~ev["artifact_mask"]]

    _, pupil = clean_pupil(rec["pupil"], blinks, t.min(), t.max())

    fix = ev["fixation_durations_s"]
    sac = ev["saccade_amplitudes_deg"]
    gaze_time_s = float(finite.sum()) / GAZE_GRID_HZ
    signals = {
        "pupil": pupil[np.isfinite(pupil)],
        "gaze_x": az,
        "gaze_y": el,
        "gaze_speed": speed,
        "fixation_duration": fix * 1000.0,
        "saccade_amplitude": sac,
        "saccade_peak_velocity": ev["saccade_peak_velocities"],
        "saccade_mean_velocity": ev["saccade_mean_velocities"],
        "saccade_duration": ev["saccade_durations_s"] * 1000.0,
        "blink_duration": (blinks["end"] - blinks["start"]).to_numpy(float) * 1000.0,
    }
    rates = {
        "blink_rate": len(blinks) * 60.0 / vt if vt > 0 else float("nan"),
        "fixation_rate": len(fix) / gaze_time_s if gaze_time_s > 0 else float("nan"),
        "saccade_rate": len(sac) / gaze_time_s if gaze_time_s > 0 else float("nan"),
    }
    qc = {
        "valid_time_s": vt,
        "gaze_time_s": gaze_time_s,
        "binocular_fraction": binocular_fraction(gaze),
        "refits": count_refits(rec["pupil"]),
        "n_blinks": int(len(blinks)),
        "n_fixations": int(len(fix)),
        "n_saccades": int(len(sac)),
        "fixation_median_ms": float(np.median(fix) * 1000.0) if len(fix) else float("nan"),
        "saccade_mean_velocity": _mean(ev["saccade_mean_velocities"]),
        "saccade_duration_ms": _mean(ev["saccade_durations_s"]) * 1000.0,
        "gaze_valid_fraction": float(finite.mean()) if len(xyz) else float("nan"),
        "pupil_valid_fraction": float(np.isfinite(pupil).mean()) if len(pupil) else float("nan"),
        "artifact_samples": int(ev["artifact_mask"].sum()),
    }
    return {"signals": signals, "rates": rates, "qc": qc}

def detect_events(xyz: np.ndarray, hz: float) -> dict:
    speed = angular_speed_deg_s(xyz, hz)
    labels = np.full(len(speed), np.nan)
    finite = np.isfinite(speed)
    labels[finite & (speed < IVT_THRESHOLD_DEG_S)] = _FIX
    labels[finite & (speed >= IVT_THRESHOLD_DEG_S)] = _SAC
    artifact = finite & (speed > MAX_VELOCITY_DEG_S)
    labels[artifact] = _BAD

    fix, amp, peak, mean_v, sac_dur = [], [], [], [], []
    for lab, start, stop in _runs(labels):
        if lab == _FIX:
            dur = (stop - start) / hz
            if dur >= MIN_FIXATION_S:
                fix.append(dur)
        elif lab == _SAC:
            before = labels[start - 1] if start > 0 else np.nan
            after = labels[stop] if stop < len(labels) else np.nan
            if before == _BAD or after == _BAD:
                continue
            a, b = xyz[max(start - 1, 0)], xyz[min(stop, len(xyz) - 1)]
            if not (np.isfinite(a).all() and np.isfinite(b).all()):
                continue
            amp.append(_angle_deg(a, b))
            peak.append(float(np.nanmax(speed[start:stop])))
            mean_v.append(float(np.nanmean(speed[start:stop])))
            sac_dur.append((stop - start) / hz)
    return {
        "fixation_durations_s": np.array(fix),
        "saccade_amplitudes_deg": np.array(amp),
        "saccade_peak_velocities": np.array(peak),
        "saccade_mean_velocities": np.array(mean_v),
        "saccade_durations_s": np.array(sac_dur),
        "artifact_mask": artifact,
    }

def summarize(values: np.ndarray, skip: tuple[str, ...] = ()) -> dict[str, float]:
    """The 17 statistics of one signal (minus `skip`), NaN where n is too small (module doc)."""
    x = np.asarray(values, dtype=float)
    n = len(x)
    nan = float("nan")
    out = dict.fromkeys(STATISTICS, nan)
    if n:
        mean = float(np.mean(x))
        pct = np.percentile(x, [25, 75, *_PERCENTILES.values()])
        out.update(min=float(x.min()), max=float(x.max()), range=float(x.max() - x.min()),
                   mean=mean, median=float(np.median(x)))
        out.update({k: float(v) for k, v in zip(_PERCENTILES, pct[2:])})
        if n >= 2:
            # Values equal up to float rounding (e.g. 12.5 vs 12.500000000000004) count as
            # constant: SD 0 and no skew/kurt, instead of a 1e-17 SD and a scipy warning.
            varies = not np.isclose(x.max(), x.min(), rtol=1e-12, atol=0.0)
            sd = float(np.std(x, ddof=1)) if varies else 0.0
            out.update(sd=sd, var=sd ** 2, iqr=float(pct[1] - pct[0]) if varies else 0.0,
                       cv=sd / mean if mean != 0 else nan)
            if varies and n >= 3:
                out["skew"] = float(skew(x, bias=False))
            if varies and n >= 4:
                out["kurt"] = float(kurtosis(x, fisher=True, bias=False))
    return {k: v for k, v in out.items() if k not in skip}

def flat_features(sig: dict) -> dict[str, float]:
    """The 173 features of one recording from recording_signals output, in feature_names order."""
    out = {}
    for signal in SIGNALS:
        values = np.asarray(sig["signals"][signal], dtype=float)
        if len(values) < MIN_VALUES:                       # D2: too few values to describe
            values = values[:0]
        stats = summarize(values)
        out.update({f"{signal}__{k}": v for k, v in stats.items()})
    out.update({rate: float(sig["rates"][rate]) for rate in RATES})
    return out

class TopNByAUC(SelectorMixin, BaseEstimator):
    """Keep the n features whose values alone best separate the two classes.

    Score per feature: |ROC-AUC - 0.5| over the rows where that feature is not blank (0 when a
    feature has no usable rows). Ties keep the earlier column. n larger than the number of
    features keeps them all.
    """

    def __init__(self, n: int = 20):
        self.n = n

    def fit(self, X, y):
        X = validate_data(self, X, ensure_all_finite="allow-nan", dtype=float)
        y = np.asarray(y)
        scores = np.zeros(X.shape[1])
        for j in range(X.shape[1]):
            ok = np.isfinite(X[:, j])
            if len(np.unique(y[ok])) == 2:
                scores[j] = abs(roc_auc_score(y[ok], X[ok, j]) - 0.5)
        self.scores_ = scores
        keep = np.argsort(-scores, kind="stable")[: min(self.n, X.shape[1])]
        self.support_ = np.zeros(X.shape[1], bool)
        self.support_[keep] = True
        return self

    def _get_support_mask(self):
        return self.support_

    def __sklearn_tags__(self):
        tags = super().__sklearn_tags__()
        tags.input_tags.allow_nan = True
        return tags

def make_logistic_regression_topn() -> Pipeline:
    """Impute (training median) -> pick top N -> scale (training mean/SD) -> elastic-net LR."""
    return Pipeline([
        ("impute", SimpleImputer(strategy="median")),

def run_lopo(X, y, groups, factory, space, n_iter=N_ITER, tune=True, params=None) -> dict:
    X = pd.DataFrame(X).reset_index(drop=True)
    y, groups = np.asarray(y), np.asarray(groups)
    oof = np.full(len(y), np.nan)
    folds, chosen = [], []
    for train_idx, test_idx in LeaveOneGroupOut().split(X, y, groups):
        est = factory()
        if params:
            est.set_params(**params)
        if tune and space:
            inner = StratifiedGroupKFold(n_splits=INNER_FOLDS, shuffle=True, random_state=RANDOM_STATE)
            search = RandomizedSearchCV(est, space, n_iter=n_iter, scoring="roc_auc", cv=inner,
                                        random_state=RANDOM_STATE, n_jobs=SEARCH_JOBS, error_score=np.nan)
            search.fit(X.iloc[train_idx], y[train_idx], groups=groups[train_idx])
            model, best = search.best_estimator_, search.best_params_
        else:
            model, best = clone(est).fit(X.iloc[train_idx], y[train_idx]), dict(params or {})
        oof[test_idx] = model.predict_proba(X.iloc[test_idx])[:, 1]
        folds.append({"model": model, "train_idx": train_idx, "test_idx": test_idx,
                      "participant": groups[test_idx][0]})
        chosen.append(best)
    return {"oof_proba": oof, "fold_models": folds, "chosen_params": chosen}

def permutation_test(X, y, groups, factory, params, n_perm=N_PERMUTATIONS, n_jobs=None) -> dict:
    """Grouped permutation test against chance (§10).

    Labels are permuted within participants (each participant's A1/A4 swapped at random),
    keeping the paired structure; hyperparameters are fixed to `params` for speed.
    All permuted label vectors are drawn first, in a fixed order, and only the refits run
    in parallel (PERM_JOBS workers), so the p-value does not depend on the worker count.
    """
    y, groups = np.asarray(y), np.asarray(groups)
    observed = _null_auc(X, y, groups, factory, params)
    rng = np.random.default_rng(RANDOM_STATE)
    permuted = []
    for _ in range(n_perm):
        yp = y.copy()
        for g in np.unique(groups):
            if rng.random() < 0.5:
                idx = np.flatnonzero(groups == g)
                yp[idx] = yp[idx][::-1]
        permuted.append(yp)
    jobs = PERM_JOBS if n_jobs is None else n_jobs
    null = Parallel(n_jobs=jobs)(delayed(_null_auc)(X, yp, groups, factory, params) for yp in permuted)
    p = (1 + sum(n >= observed for n in null)) / (1 + n_perm)
    return {"observed_auc": float(observed), "p_value": float(p), "n_perm": int(n_perm)}

def fold_shap(fold_models, X: pd.DataFrame, kind: str) -> np.ndarray:
    """Out-of-fold SHAP over all features: each fold's model explains the participant it never
    saw; features that fold did not pick get 0 (MD4)."""
    import shap

    names = list(X.columns)
    out = np.zeros(X.shape)
    for f in fold_models:
        model, tr, te = f["model"], f["train_idx"], f["test_idx"]
        cols = [names.index(c) for c in selected_features(model, names)]
        if kind == "xgb":
            sel = model.named_steps["select"].transform(X.iloc[te])
            values = shap.TreeExplainer(model.named_steps["clf"]).shap_values(sel)
        else:
            pre = model[:-1]
            explainer = shap.LinearExplainer(model.named_steps["clf"], pre.transform(X.iloc[tr]))
            values = explainer.shap_values(pre.transform(X.iloc[te]))
        values = np.asarray(values)
        if values.ndim == 3:
            values = values[:, :, -1]
        out[np.ix_(te, cols)] = values
    return out
